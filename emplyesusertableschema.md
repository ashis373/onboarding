CREATE TABLE `user` (
  `uid` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(70) NOT NULL,
  `password` varchar(70) NOT NULL,
  `emp_code` varchar(50) NOT NULL,
  `firstname` varchar(70) NOT NULL,
  `lastname` varchar(70) NOT NULL,
  `gender` int(1) NOT NULL COMMENT '1- Male, 2- Female, 3-Other',
  `profile_img` varchar(200) NOT NULL,
  `birthdate` date NOT NULL,
  `ofc_mail` varchar(80) NOT NULL,
  `department` int(10) NOT NULL,
  `ofc_location` varchar(100) NOT NULL DEFAULT 'Bhubaneswar',
  `designation` int(10) NOT NULL,
  `joiningdate` date NOT NULL,
  `qualification` varchar(150) NOT NULL,
  `personal_email` varchar(90) NOT NULL,
  `phone` varchar(15) NOT NULL,
  `address` varchar(130) NOT NULL,
  `emp_type` varchar(80) NOT NULL DEFAULT 'Full Time' COMMENT 'Full Time | Part Time |Temporary | Internal',
  `status` int(1) NOT NULL DEFAULT 0,
  `session_key` varchar(100) DEFAULT NULL,
  `app_session_key` varchar(150) NOT NULL,
  `user_role` int(1) NOT NULL DEFAULT 4 COMMENT '1 -> Admin, 2->senior employee, 3->Team lead, 4->Employee',
  `special_privilege` int(2) NOT NULL DEFAULT 0,
  `account_manager` int(2) NOT NULL DEFAULT 0,
  `last_active_time` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  `reporting_to` varchar(80) NOT NULL,
  `last_working_day` date DEFAULT NULL,
  `exit_reason` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`uid`)
) ENGINE=InnoDB AUTO_INCREMENT=223 DEFAULT CHARSET=latin1 COLLATE=latin1_swedish_ci
